¿Por qué salían todos conectados y cómo medir en tiempo real?

¡Nuevo! Aviso al correo: Este código incluye el envío automático de un email a slotcarhome@gmail.com en el mismo instante en que el piloto abre la app, sin que tú tengas que tener este panel abierto.

Pega esta función en el archivo index.html de tu aplicación justo cuando validas la licencia del usuario (pasándole el docId y el email del piloto):

// --- PEGAR EN TU APP (index.html) AL VALIDAR LA LICENCIA ---
async function iniciarTelemetriaEnVivo(licenseDocId, pilotEmail) {
    const ref = doc(db, 'artifacts', 'app_krono', 'public', 'data', 'access_codes', licenseDocId);
    
    // 1. Detectar País automáticamente
    let countryData = { countryCode: 'ES', country: 'Desconocido' };
    try {
        const res = await fetch('https://ipapi.co/json/');
        const geo = await res.json();
        if (geo.country_code) {
            countryData = { countryCode: geo.country_code, country: geo.country_name };
        }
    } catch(e) {}

    // 2. Avisar a Firebase de que acaba de entrar
    await updateDoc(ref, {
        ...countryData,
        last_ping: new Date().toISOString(),
        current_activity: 'En la aplicación'
    });

    // 3. ENVIAR CORREO AUTOMÁTICO A slotcarhome@gmail.com (Sin bloqueo CORS)
    if (!sessionStorage.getItem('notified_admin')) {
        sessionStorage.setItem('notified_admin', 'true');
        const formData = new FormData();
        formData.append('_subject', `🏎️ Piloto Conectado: ${pilotEmail} (${countryData.country})`);
        formData.append('Piloto', pilotEmail);
        formData.append('Pais', `${countryData.country} (${countryData.countryCode})`);
        formData.append('Hora_Inicio', new Date().toLocaleString('es-ES'));
        formData.append('App', 'KronoPersonal SILVER');
        formData.append('_template', 'table');
        formData.append('_captcha', 'false');

        fetch('https://formsubmit.co/slotcarhome@gmail.com', {
            method: 'POST',
            mode: 'no-cors',
            body: formData
        }).catch(() => {});
    }

    // 4. Latido cada 60 segundos (suma 1 minuto jugado y mantiene el estado "En línea")
    setInterval(async () => {
        if (document.visibilityState === 'visible') {
            await updateDoc(ref, {
                last_ping: new Date().toISOString(),
                playtime_minutes: increment(1) // Requiere importar 'increment' de firebase/firestore
            });
        }
    }, 60000);
}
